"""Challenge installed host REVEALS with independently generated synthetic pollen.

This private diagnostic matches declared basin and spatial support, but the
oracle's inverse-square deposition law is different from DISQOVER's K kernel.
It therefore reports residuals without an ecological accuracy verdict.
"""

from __future__ import annotations

import csv
import hashlib
import json
from math import floor, isfinite
from pathlib import Path
import shutil
import sys
from typing import Any

from bijux_pollenomics.analysis.lra.api import (
    LRARefusal,
    read_lra_product,
    run_lra_request,
    validate_lra_inputs,
)
from bijux_pollenomics_dev.lra_validation import build_known_landscape_report


TAXA = ("broadleaf", "conifer", "open_land")
OMISSIONS = (
    "chronology",
    "basin_geometry",
    "parameter_domain",
    "spatial_model",
    "ppe_covariance",
)
EFFORT = 1_000_000


def _encoded(value: object) -> bytes:
    return (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()


def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _write(path: Path, value: object) -> str:
    data = _encoded(value)
    path.write_bytes(data)
    return _sha256(data)


def _counts(probabilities: list[float]) -> list[int]:
    scaled = [value * EFFORT for value in probabilities]
    counts = [floor(value) for value in scaled]
    remaining = EFFORT - sum(counts)
    if not 0 <= remaining <= len(counts):
        raise ValueError("Invalid probability closure before count allocation")
    order = sorted(range(len(counts)), key=lambda i: (-(scaled[i] - counts[i]), i))
    for index in order[:remaining]:
        counts[index] += 1
    if sum(counts) != EFFORT or any(
        abs(count / EFFORT - probability) > 1 / EFFORT + 1e-15
        for count, probability in zip(counts, probabilities, strict=True)
    ):
        raise ValueError("Largest-remainder count allocation failed")
    return counts


def _input_bundle(
    root: Path,
    *,
    oracle_bytes: bytes,
    pollen: list[float],
    counts: list[int],
    ppe: dict[str, float],
    fall_speed: dict[str, float],
    parameter_version: str,
    dependency_lock: Path,
) -> dict[str, Any]:
    root.mkdir(parents=True, exist_ok=False)
    (root / "oracle-report.json").write_bytes(oracle_bytes)
    evidence_values: dict[str, object] = {
        "raw_snapshot": {
            "profile": "known-landscape-to-host-counts-v1",
            "oracle_sha256": _sha256(oracle_bytes),
            "case_id": "homogeneous-noise-free",
            "taxa": list(TAXA),
            "pollen_fraction": dict(zip(TAXA, pollen, strict=True)),
            "integer_count_rule": "largest-remainder; equal remainders by taxon order",
            "counting_effort": EFFORT,
            "counts": dict(zip(TAXA, counts, strict=True)),
        },
        "chronology": {
            "profile": "synthetic-declared-bin-v1",
            "age_type": "calendar_bp",
            "age_support_bp": [[5400, 5600]],
            "real_chronology_claim": False,
        },
        "geometry": {
            "profile": "synthetic-oracle-basin-v1",
            "site_type": "lake",
            "basin_radius_m": 175.0,
            "basin_diameter_m": 350,
            "region_cutoff_m": 5000.0,
            "local_ring_geometry_declared": False,
        },
        "taxa": {"profile": "synthetic-oracle-taxa-v1", "taxa": list(TAXA)},
        "parameters": {
            "profile": parameter_version,
            "ppe": ppe,
            "fall_speed_m_s": fall_speed,
            "ppe_sd": dict.fromkeys(TAXA, 0.0),
        },
        "decision": {
            "profile": "synthetic-comparison-decision-v1",
            "purpose": "exploratory",
            "calendar_bp_is_fabricated_test_support": True,
            "scientific_review": "not_supplied",
            "forward_physics_match": False,
            "accuracy_threshold": None,
        },
    }
    evidence = []
    for role, value in evidence_values.items():
        relative = f"{role}.json"
        digest = _write(root / relative, value)
        evidence.append(
            {"evidence_id": role, "role": role, "path": relative, "sha256": digest}
        )
    shutil.copy2(dependency_lock, root / "uv.lock")
    evidence.append(
        {
            "evidence_id": "dependency_lock",
            "role": "dependency_lock",
            "path": "uv.lock",
            "sha256": _sha256((root / "uv.lock").read_bytes()),
        }
    )
    request: dict[str, Any] = {
        "schema_version": "lra-request.v1",
        "purpose": "exploratory",
        "data_kind": "synthetic",
        "method_profile": "disqover-0.9.14-source-compatible",
        "operation": "reveals",
        "pollen_unit": "count",
        "taxa": list(TAXA),
        "regional": {
            "samples": [
                {
                    "sample_id": "synthetic-regional-001",
                    "site_id": "synthetic-oracle-basin",
                    "raw_evidence_id": "raw_snapshot",
                    "chronology_evidence_id": "chronology",
                    "bin_id": "synthetic-bin-001",
                    "age_type": "calendar_bp",
                    "age_support_bp": [[5400, 5600]],
                    "coordinate": 1,
                    "counts": counts,
                }
            ],
            "basin": {
                "site_type": "lake",
                "diameter_m": 350,
                "geometry_evidence_id": "geometry",
            },
        },
        "parameters": [
            {
                "taxon": taxon,
                "fallspeed_m_s": fall_speed[taxon],
                "ppe": ppe[taxon],
                "ppe_sd": 0.0,
            }
            for taxon in TAXA
        ],
        "temporal_matching": {
            "rule": "declared_bin_stationarity",
            "decision_evidence_id": "decision",
            "bins": [
                {"bin_id": "synthetic-bin-001", "younger_bp": 5400, "older_bp": 5600}
            ],
        },
        "configuration": {
            "seed": 7,
            "reveals": {
                "iterations": 2,
                "stochastic": False,
                "distance_weighting": "lsm unstable",
                "region_cutoff_m": 5000.0,
            },
        },
        "evidence": evidence,
        "versions": {
            "taxonomy": "synthetic-oracle-v1",
            "parameters": parameter_version,
            "geometry": "synthetic-oracle-basin-v1",
            "chronology": "synthetic-declared-bin-v1",
            "binning": "synthetic-declared-bin-v1",
        },
        "uncertainty_omissions": list(OMISSIONS),
        "exclusions": [],
    }
    _write(root / "request.json", request)
    return request


def _csv_rows(path: Path) -> list[dict[str, str]]:
    with path.open(newline="", encoding="utf-8") as stream:
        return list(csv.DictReader(stream, strict=True))


def _run_one(
    request: dict[str, Any], *, input_root: Path, destination: Path, truth: dict[str, float]
) -> dict[str, Any]:
    validated = validate_lra_inputs(request, input_root=input_root)
    product = run_lra_request(request, input_root=input_root, destination=destination)
    verified = read_lra_product(
        product.directory, expected_manifest_sha256=product.manifest_sha256
    )
    rows = _csv_rows(product.directory / "numerical/reconstruction.csv")
    factors = _csv_rows(product.directory / "numerical/deposition_factors.csv")
    if len(rows) != 1 or len(factors) != len(TAXA):
        raise ValueError("Unexpected REVEALS output denominator")
    estimate = {taxon: float(rows[0][f"{taxon}.mean"]) / 100 for taxon in TAXA}
    if not all(isfinite(value) and 0 <= value <= 1 for value in estimate.values()):
        raise ValueError("Nonfinite or out-of-range regional cover")
    residual = {taxon: estimate[taxon] - truth[taxon] for taxon in TAXA}
    closure_error = abs(sum(estimate.values()) - 1.0)
    # REVEALS reports each percentage to 0.001, bounding total rounding error.
    rounding_bound = len(TAXA) * 0.000005 + 1e-12
    if closure_error > rounding_bound:
        raise ValueError(f"Regional closure exceeds reported rounding: {closure_error}")
    return {
        "status": "generated_exploratory",
        "input_request_sha256": validated.sha256,
        "manifest_sha256": product.manifest_sha256,
        "scientific_content_sha256": product.scientific_content_sha256,
        "reader_manifest_sha256": verified.product.manifest_sha256,
        "estimated_cover_fraction": estimate,
        "residual_fraction": residual,
        "mae_fraction": sum(abs(value) for value in residual.values()) / len(TAXA),
        "cover_sum_fraction": sum(estimate.values()),
        "closure_error_fraction": closure_error,
        "rounding_closure_bound_fraction": rounding_bound,
        "deposition_factors": {row["index"]: float(row["K"]) for row in factors},
    }


def main() -> None:
    output = Path(sys.argv[1]).resolve()
    snapshot = Path(sys.argv[2]).resolve()
    output.mkdir(parents=True, exist_ok=False)
    oracle = build_known_landscape_report()
    payload = oracle.payload()
    case = next(row for row in payload["cases"] if row["case_id"] == "homogeneous-noise-free")
    pollen = [float(case["expected_pollen_fraction"][taxon]) for taxon in TAXA]
    truth = {taxon: float(case["true_support_vegetation_fraction"][taxon]) for taxon in TAXA}
    counts = _counts(pollen)
    true_parameters = payload["true_parameters"]
    fall_speed = {taxon: float(true_parameters["fall_speed_m_s"][taxon]) for taxon in TAXA}
    correct_ppe = {taxon: float(true_parameters["ppe"][taxon]) for taxon in TAXA}
    misspecified_case = next(row for row in payload["cases"] if row["case_id"] == "misspecified-ppe")
    misspecified_ppe = {taxon: float(misspecified_case["assumed_ppe"][taxon]) for taxon in TAXA}
    dependency_lock = snapshot / "uv.lock"
    correct_root = output / "correct" / "input"
    misspecified_root = output / "misspecified" / "input"
    correct = _input_bundle(
        correct_root,
        oracle_bytes=oracle.canonical_bytes,
        pollen=pollen,
        counts=counts,
        ppe=correct_ppe,
        fall_speed=fall_speed,
        parameter_version="synthetic-true-ppe-v1",
        dependency_lock=dependency_lock,
    )
    misspecified = _input_bundle(
        misspecified_root,
        oracle_bytes=oracle.canonical_bytes,
        pollen=pollen,
        counts=counts,
        ppe=misspecified_ppe,
        fall_speed=fall_speed,
        parameter_version="synthetic-misspecified-ppe-v1",
        dependency_lock=dependency_lock,
    )
    correct_result = _run_one(
        correct,
        input_root=correct_root,
        destination=output / "correct" / "product",
        truth=truth,
    )
    misspecified_result = _run_one(
        misspecified,
        input_root=misspecified_root,
        destination=output / "misspecified" / "product",
        truth=truth,
    )
    tampered_root = output / "tampered" / "input"
    shutil.copytree(correct_root, tampered_root)
    with (tampered_root / "raw_snapshot.json").open("ab") as stream:
        stream.write(b"tampered\n")
    try:
        validate_lra_inputs(correct, input_root=tampered_root)
    except LRARefusal as exc:
        tamper = {"status": "refused", "reason_code": exc.reason_code}
    else:
        raise ValueError("Tampered raw evidence was accepted")
    if tamper["reason_code"] != "evidence_digest_mismatch":
        raise ValueError(f"Unexpected tamper refusal: {tamper}")
    report = {
        "schema_version": "known-landscape-host-reveals-challenge.v1",
        "status": "engineering_diagnostic_only",
        "source_main_commit": "70ddf5085ea8fef060f359ef0a4ddc719fb5cb68",
        "oracle_profile": payload["profile"],
        "oracle_report_sha256": oracle.sha256,
        "case_id": case["case_id"],
        "taxa": list(TAXA),
        "input_pollen_fraction": dict(zip(TAXA, pollen, strict=True)),
        "counts": dict(zip(TAXA, counts, strict=True)),
        "counting_effort": EFFORT,
        "max_count_quantization_residual_fraction": max(
            abs(count / EFFORT - probability)
            for count, probability in zip(counts, pollen, strict=True)
        ),
        "true_support_vegetation_fraction": truth,
        "spatial_support": {
            "oracle_extent_m": 5000.0,
            "host_region_cutoff_m": 5000.0,
            "oracle_basin_radius_m": 175.0,
            "host_basin_diameter_m": 350,
            "matching_kind": "declared_geometry_only",
        },
        "correct_parameters": correct_result,
        "misspecified_ppe": misspecified_result,
        "misspecified_minus_correct_mae_fraction": (
            misspecified_result["mae_fraction"] - correct_result["mae_fraction"]
        ),
        "tampered_raw_evidence": tamper,
        "scientific_limits": {
            "independent_synthetic_forward_generator": True,
            "forward_physics_match": False,
            "production_lra_recovery_validated": False,
            "real_observations_used": False,
            "empirical_ecological_accuracy": False,
            "vegetation_intervals_calibrated": False,
            "love_matched_support_evaluated": False,
            "scientifically_approved": False,
            "explanation": "Oracle area/(distance+radius)^2 times PPE and fall-speed differs from the installed REVEALS Sugita B1/DISQOVER K kernel; residuals have no predeclared pass threshold.",
        },
    }
    _write(output / "report.json", report)
    print(json.dumps({"report": str(output / "report.json"), "correct_mae_fraction": correct_result["mae_fraction"], "misspecified_mae_fraction": misspecified_result["mae_fraction"], "tamper_refusal": tamper["reason_code"]}, sort_keys=True))


if __name__ == "__main__":
    main()
