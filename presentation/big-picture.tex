\begin{frame}{What the project can answer today}
\begin{itemize}
\item Trace pollen, chronology, geography and other source records to governed outputs.
\item Reproduce selected published author calculations and locate genuine archive differences.
\item Test observed-pollen transfer for Denmark, Finland, Norway and Sweden on a declared taxon axis.
\item Run a bounded English LRA reconstruction and forward-pollen prediction study.
\end{itemize}
\scope{These are distinct evidence claims; none is an independent measured-vegetation accuracy result.}
\evidence{reports/README.md}
\end{frame}

\begin{frame}{English LRA: a real reconstruction, bounded interpretation}
\begin{itemize}
\item 15 author sites, 1250--1750 BP, 30 pollen targets after a declared transport exclusion.
\item 14 sites have three frozen forward-pollen predictions each; one fold refuses after three training optimizations fail.
\item No measured vegetation target is available for reconstruction accuracy.
\end{itemize}
\scope{This demonstrates an executable source-bound workflow, not a Nordic LRA validation.}
\evidence{docs/report/pollen-reconstruction/author-lra-constrained/README.md}
\end{frame}

\begin{frame}{Published calculation and archived product are different questions}
\begin{itemize}
\item Python matches independent author-R replay on 14,074 supported means/SE values; maximum fractional residuals $1.84\times10^{-13}$ and $7.16\times10^{-14}$.
\item The exact 100--350 BP archive window differs from the author replay by up to 9.20 and 10.55 percentage points in the two grids.
\item Only one of 25 window intervals matches exactly. Source revisions and PFT transformation remain unresolved.
\end{itemize}
\scope{Numerical correspondence is established for the selected author examples; full historical paper reconstruction is open.}
\evidence{reports/science/publication-correspondence.md}
\end{frame}

\begin{frame}{The next decisions need source evidence}
\begin{enumerate}
\item Bind Swedish and other Nordic cores to selected ages, historical basin geometry, count denominators and permitted PPE use.
\item Obtain a qualified independent vegetation reference before claiming ecological accuracy.
\item Authenticate historical paper input revisions and the archive PFT/uncertainty transformation.
\item Freeze a new cohort before viewing outcomes for independent confirmation.
\end{enumerate}
\evidence{reports/needed/README.md}
\end{frame}
